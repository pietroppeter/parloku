Parloku

A mini sudoku spelled in letters, with words hidden in it. Drag the tiles up from their stacks.

01 · SUDOKUEach of the six letters appears once in every row, every column and every 2×3 box.
02 · WORDSEach clue starts in its numbered cell and is spelled one step at a time, up, down, left or right.
03 · TILESEvery letter has a stack of six tiles. Drop one on a cell, drag it away to send it home.

Parloku is Pietro's word-flavoured mini sudoku: a 6×6 grid, six letters instead of six digits, and a few words threaded through the grid as extra clues. The puzzles come from puzzlepaw, where they are crafted in a terminal UI.

This page is a nimib notebook. Everything below is the Nim source of the game you just played. Most blocks run twice: natively while nimib builds the page (that is where the printed outputs come from) and in your browser, compiled with nim js. The dragging uses a tiny port of Declarative Dragging (Horowitz, Blinn & Heer, UIST 2026): nobody wrote a drop handler, the code only lists the boards a drag can reach.

01 · THE MODELSix letters, thirty-six tiles

A puzzle is six letters, a few given letters and the word clues. A board says which tile sits in each cell. There are 36 tiles, six copies of each letter, so tile id is copy id mod 6 of letter id div 6. A given letter is just a tile that is glued down.

Puzzles travel as a short code that fits in a URL, so a future builder (or a daily puzzle) only has to hand out a link. The code is LETTERS_GIVENS_WORDS (the format puzzlepaw uses too): givens are row, column, letter and words are row, column, word, both 0-based and separated by dots.

import std/[math, strformat, strutils, sequtils]

type
  Coord = tuple[row, col: int]
  Given = object
    pos: Coord
    letter: char
  Clue = object
    pos: Coord
    word: string
  Puzzle = object
    letters: string        # the six letters
    givens: seq[Given]
    clues: seq[Clue]
  Grid = array[36, char]   # '.' for an empty cell
  Board = object
    cells: array[36, int]  # tile id in each cell, -1 when empty
    moves: int

func idx(r, c: int): int = r * 6 + c
func idx(p: Coord): int = idx(p.row, p.col)

proc encode(p: Puzzle): string =
  ## LETTERS_GIVENS_WORDS, the code format shared with puzzlepaw
  p.letters & "_" & p.givens.mapIt(&"{it.pos.row}{it.pos.col}{it.letter}").join(".") &
    "_" & p.clues.mapIt(&"{it.pos.row}{it.pos.col}{it.word}").join(".")

proc decode(code: string): Puzzle =
  var c = code.strip(chars = {'?', '#', ' '})
  if c.startsWith("p="): c = c[2 .. ^1]
  let parts = c.toUpperAscii.split('_')
  if parts.len != 3: return
  result.letters = parts[0]
  for kind in 1 .. 2:
    for item in parts[kind].split('.'):
      if item.len >= 3 and item[0] in '0'..'5' and item[1] in '0'..'5':
        let pos = (row: ord(item[0]) - ord('0'), col: ord(item[1]) - ord('0'))
        if kind == 1: result.givens.add Given(pos: pos, letter: item[2])
        else: result.clues.add Clue(pos: pos, word: item[2 .. ^1])

proc valid(p: Puzzle): bool =
  p.letters.len == 6 and p.letters.deduplicate.len == 6 and
    p.givens.allIt(it.letter in p.letters and it.pos.row in 0..5 and it.pos.col in 0..5) and
    p.clues.allIt(it.word.allIt(it in p.letters) and it.pos.row in 0..5 and it.pos.col in 0..5)

var puz: Puzzle                # the puzzle being played

proc letterOf(id: int): char = puz.letters[id div 6]
proc isGiven(cell: int): bool = puz.givens.anyIt(idx(it.pos) == cell)

proc grid(b: Board): Grid =
  for c, id in b.cells:
    result[c] = '.'
    if id >= 0: result[c] = letterOf(id)

proc boardOf(p: Puzzle, g: Grid): Board =
  ## a board holding the letters of `g`, using the lowest free copy of each letter
  var used: array[6, int]
  for c in 0 ..< 36:
    result.cells[c] = -1
    let li = p.letters.find(g[c])
    if li >= 0 and used[li] < 6:
      result.cells[c] = li * 6 + used[li]
      inc used[li]

proc startGrid(p: Puzzle): Grid =
  for c in result.mitems: c = '.'
  for gv in p.givens: result[idx(gv.pos)] = gv.letter

02 · THE RULES, AS CODEChecking, tracing, solving

Three functions carry the rules. clashes finds repeated letters in a row, column or box (they glow red on the board). trace walks a clue through the grid with a depth-first search and returns the longest prefix it can spell, so the clue chips light up letter by letter and the path is drawn over the tiles. With wild = true an empty cell matches any letter, which is what a solver needs to prune early.

proc makeUnits(): seq[seq[int]] =
  for r in 0 .. 5: result.add toSeq(0 .. 5).mapIt(idx(r, it))
  for c in 0 .. 5: result.add toSeq(0 .. 5).mapIt(idx(it, c))
  for b in 0 .. 5:
    var u: seq[int]
    for r in 0 .. 1:
      for c in 0 .. 2: u.add idx((b div 2) * 2 + r, (b mod 2) * 3 + c)
    result.add u
let units = makeUnits()

proc peers(cell: int): seq[int] =
  for u in units:
    if cell in u:
      for j in u:
        if j != cell and j notin result: result.add j
let peersOf = toSeq(0 ..< 36).mapIt(peers(it))

proc clashes(g: Grid): seq[int] =
  for c in 0 ..< 36:
    if g[c] != '.' and peersOf[c].anyIt(g[it] == g[c]): result.add c

proc trace(g: Grid, clue: Clue, wild = false): seq[int] =
  ## longest prefix of the word that can be walked from its start cell (cells not reused)
  var best, path: seq[int]
  let word = clue.word
  proc walk(cell, i: int) =
    if best.len == word.len or cell in path: return
    let ok = g[cell] == word[i] or (wild and g[cell] == '.')
    if not ok: return
    path.add cell
    if path.len > best.len: best = path
    if i + 1 < word.len:
      let (r, c) = (cell div 6, cell mod 6)
      if r > 0: walk(cell - 6, i + 1)
      if r < 5: walk(cell + 6, i + 1)
      if c > 0: walk(cell - 1, i + 1)
      if c < 5: walk(cell + 1, i + 1)
    discard path.pop
  walk(idx(clue.pos), 0)
  best

proc solved(p: Puzzle, g: Grid): bool =
  '.' notin g and g.clashes.len == 0 and p.clues.allIt(g.trace(it).len == it.word.len)

proc solve(p: Puzzle, limit = 2): seq[Grid] =
  ## backtracking with "fewest candidates first", pruning on the clues;
  ## stops after `limit` solutions (2 is enough to tell unique from ambiguous)
  var g = startGrid(p)
  var sols: seq[Grid]
  proc fits(g: Grid, cell: int, ch: char): bool = peersOf[cell].allIt(g[it] != ch)
  proc cluesOk(g: Grid): bool = p.clues.allIt(g.trace(it, wild = true).len == it.word.len)
  proc go(g: var Grid) =
    if sols.len >= limit: return
    var best = -1
    var bestN = 99
    for c in 0 ..< 36:
      if g[c] == '.':
        let n = p.letters.countIt(fits(g, c, it))
        if n < bestN: (best, bestN) = (c, n)
    if best < 0:
      if solved(p, g): sols.add g
      return
    for ch in p.letters:
      if fits(g, best, ch):
        g[best] = ch
        if cluesOk(g): go(g)
        g[best] = '.'
  if g.clashes.len == 0 and cluesOk(g): go(g)
  sols

03 · THE DRAG SPECListing futures instead of handling drops

The drag engine is the one from Declarative Dragging, in Nim: a view is a pure function from model to keyed, positioned items, and a behaviour turns a list of reachable models into a picker that chooses the one landing the held item closest to the pointer.

type
  Vec = object
    x, y: float
  Item = object            ## one keyed element of a rendered view
    key: string            ## identity across states (what your eye tracks)
    pos: Vec               ## where it sits in this state
    svg: string            ## what it looks like, drawn around (0, 0)
  Render[S] = proc (s: S): seq[Item]
  Choice[S] = object
    ok: bool
    state: S
    land: Vec              ## where the dragged element lands in `state`
    dist: float
  Picker[S] = object       ## a behaviour, instantiated at drag start
    lands: seq[Vec]        ## speculative landing spots
    pick: proc (p: Vec): Choice[S]
  Behavior[S] = proc (render: Render[S], key: string): Picker[S]

func vec(x, y: float): Vec = Vec(x: x, y: y)
func `+`(a, b: Vec): Vec = vec(a.x + b.x, a.y + b.y)
func `-`(a, b: Vec): Vec = vec(a.x - b.x, a.y - b.y)
func norm(a: Vec): float = hypot(a.x, a.y)

proc locate[S](render: Render[S], s: S, key: string): Vec =
  ## speculative rendering: draw `s` without painting it, report where `key` lands
  for it in render(s):
    if it.key == key: return it.pos
  vec(NaN, NaN)

proc closest[S](states: seq[S]): Behavior[S] =
  result = proc (render: Render[S], key: string): Picker[S] =
    var lands: seq[Vec]
    for s in states: lands.add render.locate(s, key)
    result.lands = lands
    result.pick = proc (p: Vec): Choice[S] =
      result.dist = Inf
      for i, s in states:
        let d = norm(lands[i] - p)
        if d < result.dist:
          result = Choice[S](ok: true, state: s, land: lands[i], dist: d)

proc within[S](b: Behavior[S], radius: float): Behavior[S] =
  result = proc (render: Render[S], key: string): Picker[S] =
    let inner = b(render, key)
    result.lands = inner.lands
    result.pick = proc (p: Vec): Choice[S] =
      result = inner.pick(p)
      if result.dist > radius: result.ok = false

proc orElse[S](a, b: Behavior[S]): Behavior[S] =
  result = proc (render: Render[S], key: string): Picker[S] =
    let pa = a(render, key)
    let pb = b(render, key)
    result.lands = pa.lands & pb.lands
    result.pick = proc (p: Vec): Choice[S] =
      result = pa.pick(p)
      if not result.ok: result = pb.pick(p)

The view puts each tile either in its cell or on its letter's stack. All the tiles of one stack sit at the same spot, so the stack is drawn as layers in the backdrop, with a badge counting the tiles still to place. Six glazes, one per letter, in the colours of Italian maiolica.

proc renderBoard(b: Board): seq[Item] =
  var at: array[36, int]
  for id in 0 ..< 36: at[id] = -1
  for c, id in b.cells:
    if id >= 0: at[id] = c
  for id in 0 ..< 36:
    let c = at[id]
    let pos = if c >= 0: cellPos(c) else: stackPos(id div 6)
    result.add Item(key: &"t{id}", pos: pos, svg: tileSvg(id div 6, c >= 0 and isGiven(c)))

And here is the entire interaction. Grabbing tile id lists one future per free cell (the tile in that cell; if it came from another cell, the tiles swap; otherwise the old occupant goes home), snaps to the nearest within half a cell, or else falls back to "the tile went back to its stack". Givens have no spec, so they can't be picked up.

proc place(b: Board, id, cell: int): Board =
  result = b
  let src = b.cells.find(id)
  let other = b.cells[cell]
  if src == cell: return
  inc result.moves
  if src >= 0: result.cells[src] = -1
  result.cells[cell] = id
  if other >= 0 and src >= 0: result.cells[src] = other   # swap

proc home(b: Board, id: int): Board =
  result = b
  let src = b.cells.find(id)
  if src >= 0:
    result.cells[src] = -1
    inc result.moves

proc tileSpec(b: Board, key: string): Behavior[Board] =
  let id = parseInt(key[1 .. ^1])
  let src = b.cells.find(id)
  if src >= 0 and isGiven(src): return nil
  var futures: seq[Board]
  for c in 0 ..< 36:
    if not isGiven(c): futures.add b.place(id, c)
  closest(futures).within(cell * 0.6).orElse(closest(@[b.home(id)]))

04 · THE RUNTIMEPointer events in, pick calls out

The browser runtime (std/dom) keeps one SVG group per tile key, so a CSS transition slides tiles to their new places whenever the shown board changes. While a tile is held it floats under the pointer, everything else shows the picked future, and on release that future is committed.

nbText: """
Three functions carry the rules. `clashes` finds repeated letters in a row, column or box (they glow red on the board).
`trace` walks a clue through the grid with a depth-first search and returns the longest prefix it can spell, so

  proc makeUnits(): seq[seq[int]] =
    for r in 0 .. 5: result.add toSeq(0 .. 5).mapIt(idx(r, it))
    for c in 0 .. 5: result.add toSeq(0 .. 5).mapIt(idx(it, c))
    for b in 0 .. 5:
      var u: seq[int]
      for r in 0 .. 1:
        for c in 0 .. 2: u.add idx((b div 2) * 2 + r, (b mod 2) * 3 + c)
      result.add u
  let units = makeUnits()

  proc peers(cell: int): seq[int] =
    for u in units:
      if cell in u:
        for j in u:
          if j != cell and j notin result: result.add j
  let peersOf = toSeq(0 ..< 36).mapIt(peers(it))

  proc clashes(g: Grid): seq[int] =
    for c in 0 ..< 36:
      if g[c] != '.' and peersOf[c].anyIt(g[it] == g[c]): result.add c

  proc trace(g: Grid, clue: Clue, wild = false): seq[int] =
    ## longest prefix of the word that can be walked from its start cell (cells not reused)
    var best, path: seq[int]
    let word = clue.word
    proc walk(cell, i: int) =
      if best.len == word.len or cell in path: return
      let ok = g[cell] == word[i] or (wild and g[cell] == '.')
      if not ok: return
      path.add cell
      if path.len > best.len: best = path
      if i + 1 < word.len:
        let (r, c) = (cell div 6, cell mod 6)
        if r > 0: walk(cell - 6, i + 1)
        if r < 5: walk(cell + 6, i + 1)
        if c > 0: walk(cell - 1, i + 1)
        if c < 5: walk(cell + 1, i + 1)
      discard path.pop
    walk(idx(clue.pos), 0)
    best

  proc solved(p: Puzzle, g: Grid): bool =
    '.' notin g and g.clashes.len == 0 and p.clues.allIt(g.trace(it).len == it.word.len)

  proc solve(p: Puzzle, limit = 2): seq[Grid] =
    ## backtracking with "fewest candidates first", pruning on the clues;
    ## stops after `limit` solutions (2 is enough to tell unique from ambiguous)
    var g = startGrid(p)
    var sols: seq[Grid]
    proc fits(g: Grid, cell: int, ch: char): bool = peersOf[cell].allIt(g[it] != ch)
    proc cluesOk(g: Grid): bool = p.clues.allIt(g.trace(it, wild = true).len == it.word.len)
    proc go(g: var Grid) =
      if sols.len >= limit: return
      var best = -1
      var bestN = 99
      for c in 0 ..< 36:
        if g[c] == '.':
          let n = p.letters.countIt(fits(g, c, it))
          if n < bestN: (best, bestN) = (c, n)
      if best < 0:
        if solved(p, g): sols.add g
        return
      for ch in p.letters:
        if fits(g, best, ch):
          g[best] = ch
          if cluesOk(g): go(g)
          g[best] = '.'
    if g.clashes.len == 0 and cluesOk(g): go(g)
    sols

      pos: Vec               ## where it sits in this state
      svg: string            ## what it looks like, drawn around (0, 0)

      state: S
      land: Vec              ## where the dragged element lands in `state`
      dist: float
    Picker[S] = object       ## a behaviour, instantiated at drag start

      pick: proc (p: Vec): Choice[S]

  func vec(x, y: float): Vec = Vec(x: x, y: y)

  func `-`(a, b: Vec): Vec = vec(a.x - b.x, a.y - b.y)

      result.lands = lands

        result.dist = Inf
        for i, s in states:

          if d < result.dist:
