GARDEN DEVLOG · 01

Tiles that divide like cells

In the first Parloku every letter came as a stack of six tiles. Here we try something else: a tile is a cell. Pull a new one out of it, or push two back together into one.

01 · ONE CELLDivide it, then put it back together

Here's a single cell, without a letter. Drag it to the right. A bud swells and stretches on a neck that thins as you go. Near an empty ring the bud splits away; come back and it joins again. Let go on a ring and you have two cells; let go anywhere else and the bud is pulled home.

This cell may make one copy. Once it has, the copy can't divide, so it comes away whole: move it to the other ring, or drop it back on its mother and the two coalesce into one.

02 · THE ANIMATIONTwo circles, a neck and a spring

Everything you just saw is drawn with two circles and a neck between them. The tiles zogyn moves are only letters and invisible hit areas; the soft bodies are a separate layer underneath. A dividing cell is two circles (mother and bud) and a neck. The neck is a closed path. On each side a curve leaves the mother along its outline, dips to a waist and climbs onto the bud. All the character is in one number: the waist's half-width, as a function of how far apart the two centres are. It starts fat (a bulge), thins fast, then slowly, and never quite reaches zero. It's a thread that only breaks when the cell has divided (what decides that comes later).

import std/[strformat, strutils, math, sequtils]
import zogyn
const ink = "#24361f"
const glazes = {'*': "#c3e39b", 'S': "#ffe08a", 'O': "#ffb38a", 'L': "#9fd8cb", 'A': "#ffb38a", 'I': "#c3e39b",
                'P': "#f7a8bd", 'T': "#b9c9f3"}

func glaze(c: char): string =
  for (k, v) in glazes:
    if k == c: return v
  "#e6e4cc"

func smoothstep(a, b, x: float): float =
  let t = clamp((x - a) / (b - a), 0.0, 1.0)
  t * t * (3 - 2 * t)

func waist(d, r: float): float =
  ## half-width of the neck between cells of radius r whose centres are d apart
  let t = clamp((d - 0.4 * r) / (4.2 * r), 0.0, 1.0)
  r * (0.85 - 0.74 * t)

func neck(c1: Vec, r1: float, c2: Vec, r2: float, hw: float): string =
  ## SVG path joining two circles with a waist of half-width hw
  let d = dist(c1, c2)
  if d < 0.2 * r1 or hw < 0.3: return ""
  let u = (c2 - c1) * (1 / d)
  let n = vec(-u.y, u.x)
  # the neck meets each body wider than the waist: a fillet, as when a soap bubble pinches
  let a1 = arcsin(min(0.94, 1.05 * hw / r1 + 0.32))
  let a2 = arcsin(min(0.94, 1.05 * hw / r2 + 0.32))
  let m = c1 + u * (r1 + (d - r1 - r2) / 2)
  func f(v: Vec): string = &"{v.x:.1f} {v.y:.1f}"
  var side: array[2, string]
  for k, s in [1.0, -1.0]:
    let p1 = c1 + (u * cos(a1) + n * (s * sin(a1))) * r1
    let t1 = u * sin(a1) - n * (s * cos(a1))
    let p2 = c2 + (u * -cos(a2) + n * (s * sin(a2))) * r2
    let t2 = u * -sin(a2) - n * (s * cos(a2))
    let w = m + n * (s * hw)
    # handles: along the outline only as far as needed to drop to the waist, along the axis at the waist
    let h1 = clamp(1.4 * (r1 * sin(a1) - hw) / max(cos(a1), 0.2), 1.0, dist(p1, w) * 0.5)
    let h2 = clamp(1.4 * (r2 * sin(a2) - hw) / max(cos(a2), 0.2), 1.0, dist(p2, w) * 0.5)
    let (l1, l2) = (dist(p1, w) * 0.5, dist(p2, w) * 0.5)
    side[k] = if k == 0:
      &"M{f(c1)}L{f(p1)}C{f(p1 + t1 * h1)} {f(w - u * l1)} {f(w)}C{f(w + u * l2)} {f(p2 + t2 * h2)} {f(p2)}L{f(c2)}"
    else:
      &"L{f(p2)}C{f(p2 + t2 * h2)} {f(w + u * l2)} {f(w)}C{f(w - u * l1)} {f(p1 + t1 * h1)} {f(p1)}Z"
  side[0] & side[1]

type Body = object
  at: Vec
  r: float
  letter: char
  neckTo: int        # index of the body it's joined to, or -1
  hw: float          # waist of that neck
  faded: bool        # a spent seed: drawn hollow

func membrane(bodies: openArray[Body], sw = 3.0): string =
  ## outline first (every shape stroked twice as wide), fills on top:
  ## the inner strokes are covered, only the outline of the union shows
  var ink2, fill: string
  for b in bodies:
    let c = &"""cx="{b.at.x:.1f}" cy="{b.at.y:.1f}" r="{b.r:.1f}""""
    ink2.add &"""<circle {c}/>"""
    var hue = glaze(b.letter)
    if b.faded: hue = "#f2f0e1"
    fill.add &"""<circle {c} fill="{hue}"/>"""
    if b.neckTo >= 0:
      let o = bodies[b.neckTo]
      let p = neck(b.at, b.r, o.at, o.r, b.hw)
      ink2.add &"""<path d="{p}"/>"""
      fill.add &"""<path d="{p}" fill="{glaze(b.letter)}"/>"""
  for b in bodies:           # a glint, so they read as soft and round
    fill.add &"""<ellipse cx="{b.at.x - b.r * 0.36:.1f}" cy="{b.at.y - b.r * 0.42:.1f}" rx="{b.r * 0.3:.1f}" ry="{b.r * 0.17:.1f}" fill="#fff" opacity=".55" transform="rotate(-30 {b.at.x - b.r * 0.36:.1f} {b.at.y - b.r * 0.42:.1f})"/>"""
  &"""<g fill="{ink}" stroke="{ink}" stroke-width="{2 * sw}" stroke-linejoin="round">{ink2}</g><g>{fill}</g>"""

func glyph(letter: char, r: float): string =
  if letter == '*': return ""        # the plain cell has no letter
  &"""<text y="{r * 0.36:.1f}" text-anchor="middle" font-size="{r * 1.05:.1f}" fill="{ink}">{letter}</text>"""

The outline is the old sign-painter's trick: draw every shape once with a stroke twice as thick as you want, then fill them all on top. Where shapes overlap the fills hide the inner strokes, so only the outline of the union is left. Here is the waist at work, drawn while this page was built. The mother stays put and the bud is pulled to the right. Read it right to left and it's two cells coalescing:

var strip = """<div class="strip">"""
let r = 22.0
for (d, label) in [(0.0, "at rest"), (14.0, "budding"), (34.0, "pulled"), (60.0, "stretched"),
                   (88.0, "a thread"), (88.0, "divided")]:
  let joined = label != "divided"
  let k = if joined: 1 - 0.2 * smoothstep(0.0, 2.5 * r, d) else: 1.0
  let mother = Body(at: vec(30, 40), r: r * (1 - 0.06 * (1 - k)), letter: '*', neckTo: -1)
  var bud = Body(at: vec(30 + d, 40), r: r * k, letter: '*', neckTo: -1)
  if joined: (bud.neckTo, bud.hw) = (0, waist(d, r))
  strip.add &"""<figure><svg viewBox="0 0 150 80">{membrane([mother, bud], 2.4)}</svg><figcaption>{label}</figcaption></figure>"""
strip.add "</div>"

The bud also shrinks a little while it's stretched (1 - 0.2·smoothstep), and a letter on it shrinks with it, as if it were still drawing material from its mother. Once it divides it springs back to full size.

Making it move

The drawing above is a pure function of where the two centres are. To animate it, a requestAnimationFrame loop redraws the layer while anything moves. It reads each tile's actual position from its computed CSS transform, so the bodies follow the tiles even while they glide (a CSS transition) back home or into a ring. On top of that come three small touches:

  • A spring when it splits. Mother and daughter wobble for half a second: a damped sine on their radius, 1 + 0.09·e^(-t/140)·sin(t/26) with t in milliseconds.
  • A neck that regrows. When the bud rejoins, the neck grows back from nothing over 160 ms with a little overshoot, so a quick back and forth near a ring reads as one hesitant cell, not as flicker.
  • Coalescing is budding backwards. As a free cell nears its twin, a neck reaches out between them and gets fatter as they get closer, the same waist curve with the distance shrinking. When it has glided in, the twin takes it in with a wobble.

Tiles slide with a slightly springy easing (cubic-bezier(.2,.9,.25,1.15)), so a bud released far away is pulled home and overshoots a hair before it settles.

03 · WITH LETTERSA garden, then a board

A garden of three letters

Now the cells carry letters. Drag S or O towards one of the empty rings.

Each letter may be used twice, and every tile in the garden is a cell, so the ones you planted can bud as well. L already has both of its copies out, so its seed looks empty and gives nothing. A tile that can't divide any more comes away whole instead: move it to another ring, or drop it on another L and the two fuse back into one cell. Fuse an L into its seed and the seed fills up again, ready to bud.

04 · WHAT DECIDESStates, and zogyn

So far we've only drawn. Something still has to decide what happens: does this pull make a copy, where can it go, when has the cell divided? The effect has two independent halves:

  1. Where can the copy go? That's a question about the model, and zogyn answers it. We never write a drop handler. For the grabbed tile we list the states the drag can reach: "nothing happened" plus "a copy planted in place i", one per place. A tile that can't divide lists "moved to place i" and "fused into tile k" instead. zogyn shows whichever lands the tile closest to the pointer.
  2. What does it look like? The membrane layer from the previous section. It never decides anything. Every frame it looks at which state zogyn picked and where the tiles are on screen.

The trick that ties them together: the neck exists exactly when zogyn's pick is the "nothing happened" state. If the pick is a planted state, the cell has divided. Fusing is the same picture run backwards: a neck appears when the pick is "fused into k", between the held tile and k. So "splits near a place, rejoins near home, fuses near a sibling" isn't extra code. It falls out of the same choice that decides where the tile lands.

The model: a tile leaves a fresh copy behind

Every tile has an id, and the id is its key on screen (what your eye tracks). A source is a seed or any planted tile. When you plant from a source, the tile you hold keeps its id and moves to the place, and the source gets a fresh id. So the thing in your hand is the one that moves, and a new cell appears where it was. A tile that fuses is kept for a moment in absorbed, drawn on top of its partner, so zogyn can see where it lands. Once it has glided in, the page drops it (tidy). This block runs natively (the figures below are drawn at build time) and in the page:

type
  Patch = object         # a garden: seeds that never move, places that fill up
    seeds: string        # letter of each seed
    seedId: seq[int]     # id of the tile sitting on each seed
    cell: string         # letter in each place, ' ' when empty
    cellId: seq[int]     # id of the tile in each place (0 when empty)
    absorbed: seq[(int, int)]  # (id, id it fused into): gliding into a sibling, until `tidy`
    next: int            # next fresh id
  Rule = proc (s: Patch, letter: char, place: int): bool

func tkey(id: int): string = "t" & $id

func source(s: Patch, key: string): (bool, int) =
  ## where the tile `key` sits: (true, seed j) or (false, place i)
  for j, id in s.seedId:
    if tkey(id) == key: return (true, j)
  for i, id in s.cellId:
    if id > 0 and tkey(id) == key: return (false, i)
  (false, -1)

func letterOf(s: Patch, key: string): char =
  let (seed, i) = s.source(key)
  if i < 0: '\0' elif seed: s.seeds[i] else: s.cell[i]

func plant(s: Patch, key: string, place: int): Patch =
  ## the held tile goes to `place`; its source gets a fresh tile
  result = s
  let (seed, i) = s.source(key)
  let id = if seed: s.seedId[i] else: s.cellId[i]
  result.cell[place] = s.letterOf(key)
  result.cellId[place] = id
  if seed: result.seedId[i] = s.next else: result.cellId[i] = s.next
  result.next = s.next + 1

func looksLike(s: Patch, key: string): char =
  ## the letter on screen, fusing tiles included
  result = s.letterOf(key)
  for (id, into) in s.absorbed:
    if tkey(id) == key: return s.letterOf(tkey(into))

func tidy(s: Patch): Patch =
  result = s
  result.absorbed = @[]

proc targets(s: Patch, key: string, rule: Rule): seq[Patch] =
  ## the copies the tile can bud: one state per place a copy may go
  let letter = s.letterOf(key)
  for place, c in s.cell:
    if c == ' ' and rule(s, letter, place): result.add s.plant(key, place)

proc wholeMoves(s: Patch, key: string, rule: Rule): seq[Patch] =
  ## a tile that can't divide comes away whole: moved to another place, or fused into a sibling
  let (seed, i) = s.source(key)
  if seed or i < 0: return                 # seeds never move
  let letter = s.cell[i]
  var lifted = s
  lifted.cell[i] = ' '
  lifted.cellId[i] = 0
  for place, c in lifted.cell:
    if c == ' ' and place != i and rule(lifted, letter, place):
      var moved = lifted
      moved.cell[place] = letter
      moved.cellId[place] = s.cellId[i]
      result.add moved
  for j, id in s.seedId:
    if s.seeds[j] == letter:
      var fused = lifted
      fused.absorbed.add (s.cellId[i], id)
      result.add fused
  for k, id in s.cellId:
    if k != i and s.cell[k] == letter:
      var fused = lifted
      fused.absorbed.add (s.cellId[i], id)
      result.add fused

proc cellSpec(rule: Rule, snap: float): Spec[Patch] =
  result = proc (s: Patch, key: string): Behavior[Patch] =
    if s.letterOf(key) == '\0': return nil
    let buds = s.targets(key, rule)        # it can divide: bud a copy
    if buds.len > 0:
      return closest(buds).within(snap).orElse(closest(@[s]))
    let whole = s.wholeMoves(key, rule)    # it can't: move it, or fuse it into a sibling
    if whole.len == 0: return nil          # a spent seed gives nothing
    let fuse = whole.filterIt(it.absorbed.len > 0)
    let move = whole.filterIt(it.absorbed.len == 0)
    # siblings attract from further away than empty places do
    closest(fuse).within(2 * snap).orElse(closest(move).within(snap)).orElse(closest(@[s]))

That last proc is the whole interaction. closest(places).within(snap) picks a reachable state once the tile is within snap of where it would land; .orElse(closest(@[s])) falls back to home anywhere else. Which list it gets decides the gesture: buds while the letter can still divide, whole moves and fusions once it can't (fusions are tried first and from twice as far, so a sibling seems to pull). A spent seed gets nil, so it can't even be grabbed. Nothing else in the code knows which letters are spent.

  s.cell.count(letter) < 1               # the plain cell: one copy at a time

  s.cell.count(letter) < copies

let sudokuRule: Rule = proc (s: Patch, letter: char, place: int): bool =
  let (r, c) = (place div 6, place mod 6)
  for k in 0 ..< 36:
    if s.cell[k] == letter and (k div 6 == r or k mod 6 == c or
        (k div 12 == r div 2 and (k mod 6) div 3 == c div 3)): return false
  true

Where the tiles are: one render, three layouts

zogyn needs a render: a pure function from the model to keyed, positioned items. Each item is only a letter and an invisible hit circle. The bodies are the membrane's job. The single cell has two places in a row, the garden puts its places on an arc and the board on a 6×6 grid, so they get three layouts of the same render:

type Layout = object
  r: float
  seedAt: proc (j, n: int): Vec
  placeAt: proc (i, n: int): Vec

proc cellRender(lay: Layout): Render[Patch] =
  result = proc (s: Patch): seq[Item] =
    let tile = proc (c: char): string =
      &"""<circle r="{lay.r}" fill="transparent"/><g class="nuc">{glyph(c, lay.r)}</g>"""
    for j, id in s.seedId:
      result.add Item(key: tkey(id), pos: lay.seedAt(j, s.seeds.len), svg: tile(s.seeds[j]))
    for i, id in s.cellId:
      if id > 0:
        result.add Item(key: tkey(id), pos: lay.placeAt(i, s.cell.len), svg: tile(s.cell[i]))
    for (id, into) in s.absorbed:          # a fusing tile lands on its partner
      let (seed, k) = s.source(tkey(into))
      var at = lay.placeAt(k, s.cell.len)
      if seed: at = lay.seedAt(k, s.seeds.len)
      result.add Item(key: tkey(id), pos: at, svg: tile(s.looksLike(tkey(id))))

The effect layer: drawing what zogyn picked

Back to the animation loop, now that we know what zogyn offers. Each frame it answers three questions from zogyn's pick:

  • Is the held tile joined? Yes when zogyn's current pick is the committed state ("nothing happened"). Then it draws the mother at the tile's home with its letter, the bud under the pointer and the neck between them. After a release back home the tile glides back, and the neck follows it until they touch.
  • Did it just split or rejoin? When the answer flips, it starts the spring (on a split) or regrows the neck (on a rejoin).
  • Is it fusing? When the pick is "fused into k" (or a released tile is still gliding into k), it draws the neck between the tile and k. When the tile has arrived the page tidies it out of the model.

Spent seeds are drawn hollow with a faded letter. The layer finds them by asking the spec: nil means nothing to give.

None of this touches zogyn: its stage already exposes the pick (choice), the candidates (picker.lands), the held key, and onMove/onCommit hooks. The layer is about 150 lines:

  for (id, into) in s.absorbed:
    if tkey(id) == key: return tkey(into)

      if home: c.joinAt = t else: c.splitAt = t
    c.joined = buds and home
    c.tether = ""
    if c.joined: c.tether = held

  for e in st.fg.querySelectorAll("[data-key]"):
    onPage.add ($e.getAttribute("data-key"), e.onScreen)
  let homeOfTether = if c.tether.len > 0: st.render.locate(st.state, c.tether) else: nowhere
  let wob = spring(t - c.splitAt, 0.09)
  var bodies: seq[Body]
  var index: Table[string, int]
  var motherGlyph = ""
  for (key, at) in onPage:
    let letter = s.looksLike(key)
    if letter == '\0': continue

    if key == c.tether:

      else:
        let k = 1 - 0.2 * smoothstep(0.0, 2.5 * r, d)

    if buds and (key == held or dist(at, st.render.locate(st.state, held)) < 1): k = wob
    if key == c.mergeKey: k = spring(t - c.mergeAt, 0.12)
    let spent = c.hollow and st.spec(st.state, key).isNil and st.state.letterOf(key) != '\0'
    bodies.add Body(at: at, r: r * k, letter: letter, neckTo: -1, faded: spent)

    var o: cstring = "1"
    if spent: o = "0.4"
    n.querySelector(".nuc").setCss("opacity", o)
  # fusing pairs: the held tile over its chosen partner, or a released one gliding in
  var pairs: seq[(string, string)]
  if held.len > 0 and st.choice.ok and s.fusing(held).len > 0: pairs.add (held, s.fusing(held))

      if d > 0.6:

        bodies[ia].hw = waist(d, r)

  c.awake = max(c.awake, now() + ms)
  if c.running: return

    c.frame()
    if c.st.dragKey.len > 0 or now() < c.awake: raf(step) else: c.running = false

   Every frame it looks at which state zogyn picked and where the tiles are on screen.

If the pick is a planted state, the cell has divided. Fusing is the same picture run backwards: a neck appears when
the pick is "fused into *k*", between the held tile and *k*. So "splits near a place, rejoins near home, fuses near a
sibling" isn't extra code. It falls out of the same choice that decides where the tile lands.
"""

nbRawHtml: """<h3>The model: a tile leaves a fresh copy behind</h3>"""

nbText: """
Every tile has an id, and the id is its key on screen (what your eye tracks). A source is a seed or any planted tile.
When you plant from a source, *the tile you hold* keeps its id and moves to the place, and the source gets a **fresh id**.
So the thing in your hand is the one that moves, and a new cell appears where it was. A tile that fuses
is kept for a moment in `absorbed`, drawn on top of its partner, so zogyn can see where it lands. Once it has
glided in, the page drops it (`tidy`).
This block runs natively (the figures below are drawn at build time) and in the page:
"""

nbCodeAnd(nbJsFromCode):
  type
    Patch = object         # a garden: seeds that never move, places that fill up
      seeds: string        # letter of each seed
      seedId: seq[int]     # id of the tile sitting on each seed
      cell: string         # letter in each place, ' ' when empty
      cellId: seq[int]     # id of the tile in each place (0 when empty)
      absorbed: seq[(int, int)]  # (id, id it fused into): gliding into a sibling, until `tidy`
      next: int            # next fresh id
    Rule = proc (s: Patch, letter: char, place: int): bool

  func tkey(id: int): string = "t" & $id

  func source(s: Patch, key: string): (bool, int) =
    ## where the tile `key` sits: (true, seed j) or (false, place i)
    for j, id in s.seedId:
      if tkey(id) == key: return (true, j)
    for i, id in s.cellId:
      if id > 0 and tkey(id) == key: return (false, i)
    (false, -1)

  func letterOf(s: Patch, key: string): char =
    let (seed, i) = s.source(key)
    if i < 0: '\0' elif seed: s.seeds[i] else: s.cell[i]

  func plant(s: Patch, key: string, place: int): Patch =
    ## the held tile goes to `place`; its source gets a fresh tile
    result = s
    let (seed, i) = s.source(key)
    let id = if seed: s.seedId[i] else: s.cellId[i]
    result.cell[place] = s.letterOf(key)
    result.cellId[place] = id
    if seed: result.seedId[i] = s.next else: result.cellId[i] = s.next
    result.next = s.next + 1

  func looksLike(s: Patch, key: string): char =
    ## the letter on screen, fusing tiles included
    result = s.letterOf(key)
    for (id, into) in s.absorbed:
      if tkey(id) == key: return s.letterOf(tkey(into))

  func tidy(s: Patch): Patch =
    result = s
    result.absorbed = @[]

  proc targets(s: Patch, key: string, rule: Rule): seq[Patch] =
    ## the copies the tile can bud: one state per place a copy may go
    let letter = s.letterOf(key)
    for place, c in s.cell:
      if c == ' ' and rule(s, letter, place): result.add s.plant(key, place)

  proc wholeMoves(s: Patch, key: string, rule: Rule): seq[Patch] =
    ## a tile that can't divide comes away whole: moved to another place, or fused into a sibling
    let (seed, i) = s.source(key)
    if seed or i < 0: return                 # seeds never move
    let letter = s.cell[i]
    var lifted = s
    lifted.cell[i] = ' '
    lifted.cellId[i] = 0
    for place, c in lifted.cell:
      if c == ' ' and place != i and rule(lifted, letter, place):
        var moved = lifted
        moved.cell[place] = letter
        moved.cellId[place] = s.cellId[i]
        result.add moved
    for j, id in s.seedId:
      if s.seeds[j] == letter:
        var fused = lifted
        fused.absorbed.add (s.cellId[i], id)
        result.add fused
    for k, id in s.cellId:
      if k != i and s.cell[k] == letter:
        var fused = lifted
        fused.absorbed.add (s.cellId[i], id)
        result.add fused

  proc cellSpec(rule: Rule, snap: float): Spec[Patch] =
    result = proc (s: Patch, key: string): Behavior[Patch] =
      if s.letterOf(key) == '\0': return nil
      let buds = s.targets(key, rule)        # it can divide: bud a copy
      if buds.len > 0:
        return closest(buds).within(snap).orElse(closest(@[s]))
      let whole = s.wholeMoves(key, rule)    # it can't: move it, or fuse it into a sibling
      if whole.len == 0: return nil          # a spent seed gives nothing
      let fuse = whole.filterIt(it.absorbed.len > 0)
      let move = whole.filterIt(it.absorbed.len == 0)
      # siblings attract from further away than empty places do
      closest(fuse).within(2 * snap).orElse(closest(move).within(snap)).orElse(closest(@[s]))

nbText: """
That last proc is the whole interaction. `closest(places).within(snap)` picks a reachable state once the tile is
within `snap` of where it would land; `.orElse(closest(@[s]))` falls back to home anywhere else. Which list it gets
decides the gesture: buds while the letter can still divide, whole moves and fusions once it can't (fusions are
tried first and from twice as far, so a sibling seems to pull). A spent seed
gets `nil`, so it can't even be grabbed. Nothing else in the code knows which letters are spent.

The two rules differ in one line each:
"""

nbCodeAnd(nbJsFromCode):
  const copies = 2
  let cellRule: Rule = proc (s: Patch, letter: char, place: int): bool =
    s.cell.count(letter) < 1               # the plain cell: one copy at a time
  let gardenRule: Rule = proc (s: Patch, letter: char, place: int): bool =
    s.cell.count(letter) < copies

  let sudokuRule: Rule = proc (s: Patch, letter: char, place: int): bool =
    let (r, c) = (place div 6, place mod 6)
    for k in 0 ..< 36:
      if s.cell[k] == letter and (k div 6 == r or k mod 6 == c or

    true

and an invisible hit circle. The bodies are the membrane's job. The single cell has two places in a row, the garden

nbCodeAnd(nbJsFromCode):

    seedAt: proc (j, n: int): Vec

  proc cellRender(lay: Layout): Render[Patch] =

      result.cellId[place] = result.next

  proc newBoard(letters, givens: string): Patch =

nbRawHtml: """<h3>The effect layer: drawing what zogyn picked</h3>"""

nbText: """
Back to the animation loop, now that we know what zogyn offers. Each frame it answers three questions from
zogyn's pick:

- **Is the held tile joined?** Yes when zogyn's current pick is the committed state ("nothing happened"). Then it draws
  the mother at the tile's home with its letter, the bud under the pointer and the neck between them. After a release
  back home the tile glides back, and the neck follows it until they touch.
- **Did it just split or rejoin?** When the answer flips, it starts the spring (on a split) or regrows the neck
  (on a rejoin).
- **Is it fusing?** When the pick is "fused into *k*" (or a released tile is still gliding into *k*), it draws the
  neck between the tile and *k*. When the tile has arrived the page tidies it out of the model.

Spent seeds are drawn hollow with a faded letter. The layer finds them by asking the spec: `nil` means nothing to give.

None of this touches zogyn: its stage already exposes the pick (`choice`), the candidates (`picker.lands`), the held
key, and `onMove`/`onCommit` hooks. The layer is about 150 lines:
"""

nbRawHtml: "<details><summary>the effect layer (browser only)</summary>"
nbCodeDisplay(nbJsFromCode):
  import std/[dom, tables]
  import zogyn/stage

  proc now(): float {.importjs: "performance.now()".}
  proc raf(cb: proc ()) {.importjs: "requestAnimationFrame(#)".}
  proc later(cb: proc (), ms: int) {.importjs: "setTimeout(#, #)".}
  proc onScreen(e: Element): Vec {.importjs:
    "(m => ({x: m.e, y: m.f}))(new DOMMatrix(getComputedStyle(#).transform))".}
  proc svgG(): Element {.importjs: "document.createElementNS('http://www.w3.org/2000/svg', 'g')".}
  proc setCss(e: Element, k, v: cstring) {.importjs: "#.style.setProperty(#, #)".}

  type Cells = ref object
    st: Stage[Patch]
    r: float
    rule: Rule
    hollow: bool              # draw spent seeds hollow
    layer, status: Element
    tether: string            # key still joined to its mother: held at home, or gliding back
    joined: bool              # was the held bud joined last frame?
    splitAt, joinAt: float    # when it last split / rejoined (ms)
    mergeKey: string          # the tile that just took in a sibling, and when
    mergeAt: float
    awake: float              # keep animating until then
    running: bool

  func spring(dt, amp: float): float =    # damped wobble, 1 at rest
    if dt > 600: 1.0 else: 1 + amp * exp(-dt / 140) * sin(dt / 26)

  proc node(c: Cells, key: string): Element =
    c.st.fg.querySelector(cstring(&"""[data-key="{key}"]"""))

  proc fusing(s: Patch, key: string): string =
    ## the partner `key` is fusing into in `s`, or ""
    for (id, into) in s.absorbed:
      if tkey(id) == key: return tkey(into)

  proc frame(c: Cells) =
    let (st, r, t) = (c.st, c.r, now())
    let held = st.dragKey
    let s = st.shown
    let buds = held.len > 0 and st.state.targets(held, c.rule).len > 0   # dividing, not moving whole
    if held.len > 0:
      let home = st.choice.ok and st.choice.state == st.state
      if buds and home != c.joined:
        if home: c.joinAt = t else: c.splitAt = t
      c.joined = buds and home
      c.tether = ""
      if c.joined: c.tether = held
    var onPage: seq[(string, Vec)]
    for e in st.fg.querySelectorAll("[data-key]"):
      onPage.add ($e.getAttribute("data-key"), e.onScreen)
    let homeOfTether = if c.tether.len > 0: st.render.locate(st.state, c.tether) else: nowhere
    let wob = spring(t - c.splitAt, 0.09)
    var bodies: seq[Body]
    var index: Table[string, int]
    var motherGlyph = ""
    for (key, at) in onPage:
      let letter = s.looksLike(key)
      if letter == '\0': continue
      let n = c.node(key)
      if key == c.tether:
        let d = dist(at, homeOfTether)
        if held.len == 0 and d < 0.6:
          c.tether = ""                       # glided home: one cell again
        else:
          let k = 1 - 0.2 * smoothstep(0.0, 2.5 * r, d)
          let grow = clamp((t - c.joinAt) / 160, 0.0, 1.0)
          let g = grow * (1 + 0.25 * sin(grow * PI))   # regrow the neck with a little overshoot
          bodies.add Body(at: homeOfTether, r: r * (1 - 0.06 * (1 - k)), letter: letter, neckTo: -1)
          bodies.add Body(at: at, r: r * k, letter: letter, neckTo: bodies.len - 1, hw: waist(d, r) * g)
          index[key] = bodies.len - 1
          motherGlyph = &"""<g transform="translate({homeOfTether.x:.1f} {homeOfTether.y:.1f})">{glyph(letter, r)}</g>"""
          n.querySelector(".nuc").setCss("transform", cstring(&"scale({k:.3f})"))
          continue
      var k = 1.0
      if buds and (key == held or dist(at, st.render.locate(st.state, held)) < 1): k = wob
      if key == c.mergeKey: k = spring(t - c.mergeAt, 0.12)
      let spent = c.hollow and st.spec(st.state, key).isNil and st.state.letterOf(key) != '\0'
      bodies.add Body(at: at, r: r * k, letter: letter, neckTo: -1, faded: spent)
      index[key] = bodies.len - 1
      n.querySelector(".nuc").setCss("transform", "scale(1)")
      var o: cstring = "1"
      if spent: o = "0.4"
      n.querySelector(".nuc").setCss("opacity", o)
    # fusing pairs: the held tile over its chosen partner, or a released one gliding in
    var pairs: seq[(string, string)]
    if held.len > 0 and st.choice.ok and s.fusing(held).len > 0: pairs.add (held, s.fusing(held))
    for (id, into) in st.state.absorbed: pairs.add (tkey(id), tkey(into))
    for (a, b) in pairs:
      if a in index and b in index:
        let (ia, ib) = (index[a], index[b])
        let d = dist(bodies[ia].at, bodies[ib].at)
        if d > 0.6:
          bodies[ia].neckTo = ib
          bodies[ia].hw = waist(d, r)
    c.layer.innerHTML = cstring(membrane(bodies) & motherGlyph)

  proc kick(c: Cells, ms = 900.0) =
    c.awake = max(c.awake, now() + ms)
    if c.running: return
    c.running = true
    proc step() =
      c.frame()
      if c.st.dragKey.len > 0 or now() < c.awake: raf(step) else: c.running = false
    raf(step)

  proc tidyUp(c: Cells) =
    ## once a fused tile has glided into its partner, drop it from the model
    let st = c.st
    if st.state.absorbed.len == 0: return
    if st.dragKey.len > 0:
      later(proc () = c.tidyUp(), 200)
      return
    let (_, into) = st.state.absorbed[^1]
    c.mergeKey = tkey(into)
    c.mergeAt = now()
    st.setState(st.state.tidy)

  proc grow(st: Stage[Patch], r: float, statusId: string, rule: Rule, hollow = true): Cells =
    let c = Cells(st: st, r: r, rule: rule, hollow: hollow, layer: svgG(),
                  status: document.getElementById(cstring(statusId)))
    c.layer.setAttribute("class", "membrane")
    st.svg.insertBefore(c.layer, st.fg)
    st.onMove = proc () =
      let held = st.dragKey
      if held.len > 0:
        let letter = st.state.letterOf(held)
        var (it, twin) = ($letter, "another " & $letter)
        if letter == '*': (it, twin) = ("the cell", "its twin")
        let n = st.state.targets(held, rule).len
        var msg = &"{it} can bud a copy into {n} places"
        if n == 1: msg = &"{it} can bud a copy into 1 place"
        if n == 0: msg = &"{it} can't divide any more: move it, or drop it on {twin}"
        if st.choice.ok and st.choice.state != st.state:
          if n > 0: msg = &"{it} has split off: let go to plant it"
          elif st.choice.state.fusing(held).len > 0: msg = "let go and the two fuse into one"
          else: msg = &"let go to move {it} here"
        c.status.textContent = cstring(msg)
      else:
        c.status.textContent = ""
      c.kick()
    st.onCommit = proc (before, after: Patch) =
      if after.absorbed.len > 0: later(proc () = c.tidyUp(), 380)
    c.kick()
    c
